cmake_minimum_required(VERSION 3.22)
project(psms VERSION 0.1.8 LANGUAGES NONE)
if(NOT IS_ABSOLUTE "${BIN_DIR}" OR NOT LABEL MATCHES "^linux-(x64|arm64)$")
  message(FATAL_ERROR "Set BIN_DIR and LABEL for the native Linux build")
endif()
install(PROGRAMS "${BIN_DIR}/psms" "${BIN_DIR}/psms-desktop" DESTINATION bin)
install(FILES dev.papamo.psms.desktop DESTINATION share/applications)
install(FILES autostart.desktop DESTINATION /etc/xdg/autostart RENAME dev.papamo.psms-shortcut.desktop)
install(PROGRAMS desktop-entry DESTINATION lib/psms)
install(FILES ../../assets/psms-icon.svg DESTINATION share/icons/hicolor/scalable/apps RENAME dev.papamo.psms.svg)
install(FILES ../../LICENSE ../../README.md ../../SYNC.md DESTINATION share/doc/psms)
set(CPACK_PACKAGE_NAME "psms")
set(CPACK_PACKAGE_VENDOR "Papamo Software")
set(CPACK_PACKAGE_CONTACT "Papamo Software <https://www.papamo.dev/sms>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Self-hosted encrypted message archive and desktop inbox")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://www.papamo.dev/sms")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_SET_DESTDIR ON)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "xdg-utils, xdg-user-dirs")
set(CPACK_DEBIAN_FILE_NAME "psms-${LABEL}.deb")
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_REQUIRES "xdg-utils, xdg-user-dirs")
set(CPACK_RPM_FILE_NAME "psms-${LABEL}.rpm")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /etc/xdg /etc/xdg/autostart /usr/share/applications /usr/share/icons /usr/share/icons/hicolor /usr/share/icons/hicolor/scalable /usr/share/icons/hicolor/scalable/apps)
include(CPack)
